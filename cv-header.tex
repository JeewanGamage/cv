% Fonts: Lato (body) and Roboto (headings). If a fonts/ folder with the .ttf

  \setmainfont{Lato}[Path=fonts/,Extension=.ttf,UprightFont=*-Regular,BoldFont=*-Bold,ItalicFont=*-Italic,BoldItalicFont=*-BoldItalic,Ligatures=TeX]
  \newfontfamily\headingfont{Roboto}[Path=fonts/,Extension=.ttf,UprightFont=*-Regular,BoldFont=*-Bold,ItalicFont=*-Italic,BoldItalicFont=*-BoldItalic,Ligatures=TeX]

% Entry: left column dates, right column title/subtitle
\newcommand{\cventry}[2]{%
  \par\vspace{3pt}\noindent
  \begin{tabularx}{\textwidth}{@{}p{3.0cm}X@{}}
  {\small\color{awesome}\headingfont #1} & #2 \\
  \end{tabularx}\par\vspace{-4pt}}

% Header block
\newcommand{\cvheader}[6]{%
  % #1 name, #2 position, #3 affiliation, #4 contact line, #5 tagline, #6 photo (or empty)
  \begin{center}
  \ifx\\#6\\\else
    \begin{tikzpicture}\clip (0,0) circle (1.3cm) node {\includegraphics[width=2.6cm]{#6}};\end{tikzpicture}\\[4pt]
  \fi
  {\fontsize{28pt}{34pt}\selectfont\headinglight\color{graytext}#1}\\[4pt]
  {\small\headingfont\bfseries\color{awesome}\MakeUppercase{#2}}\\[2pt]
  {\small\color{graytext}\itshape #3}\\[6pt]
  {\footnotesize\color{darktext}#4}\\[10pt]
  \begin{minipage}{0.9\textwidth}\centering\small\itshape\color{graytext}#5\end{minipage}
  \end{center}
  \vspace{4pt}}
